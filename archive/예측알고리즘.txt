# 📌 단일 파일 처리 함수
classify_meme <- function(filepath) {
  df <- read_csv(filepath, skip = 2, col_names = c("date", "value"), show_col_types = FALSE)
  
  df <- df %>%
    mutate(
      value = ifelse(str_detect(value, "<1"), 0, as.numeric(value)),
      date = ym(paste0(date, "-01")),
      week = row_number()
    ) %>%
    filter(!is.na(value))
  
  if (nrow(df) < 8) return(tibble(file = basename(filepath), type = "데이터 부족"))
  
  max_val <- max(df$value, na.rm = TRUE)
  threshold15 <- 0.15 * max_val
  threshold80 <- 0.8 * max_val
  
  # 15% 이상 4주 이상 연속 구간 찾기
  over15 <- df$value >= threshold15
  rle15 <- rle(over15)
  start_idx <- which(rle15$lengths >= 4 & rle15$values)[1]
  
  if (is.na(start_idx)) return(tibble(file = basename(filepath), type = "15% 이상 지속 없음"))
  
  idx_start <- sum(rle15$lengths[1:(start_idx - 1)]) + 1
  df_valid <- df[idx_start:nrow(df), ]
  
  # 다시 15% 미만으로 떨어지기 전까지만 사용
  if (any(df_valid$value < threshold15)) {
    min_idx <- which(df_valid$value < threshold15)[1]
    df_valid <- df_valid[1:(min_idx - 1), ]
  }
  
  # 80% 피크 도달 확인
  if (!any(df_valid$value >= threshold80)) {
    return(tibble(file = basename(filepath), type = "80% 피크 없음"))
  }
  
  peak_idx <- which.max(df_valid$value)
  if ((peak_idx + 1) > nrow(df_valid)) {
    return(tibble(file = basename(filepath), type = "하락 판단 불가"))
  }
  
  # 피크 이후 최대 4주까지 남은 데이터 사용
  post_peak <- df_valid[(peak_idx+1):min(peak_idx+4, nrow(df_valid)), ]
  if (nrow(post_peak) < 2) {
    return(tibble(file = basename(filepath), type = "하락 판단 불가"))
  }
  
  peak_val <- df_valid$value[peak_idx]
  post_avg <- mean(post_peak$value, na.rm = TRUE)
  drop_rate <- (peak_val - post_avg) / peak_val
  
  type <- ifelse(drop_rate >= 0.5, "단기", "장기")
  
  return(tibble(file = basename(filepath), type = type))
}

# 🔁 전체 파일 반복 처리
results <- lapply(file_list, classify_meme)

# 1. 분류 결과 CSV 파일 불러오기

# 2. 분류 비율 계산
summary <- result %>%
